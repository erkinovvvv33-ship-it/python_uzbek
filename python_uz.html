class BankHisobi:
    """Bank hisobini ifodalovchi class."""

    # Class o'zgaruvchisi — barcha ob'ektlar uchun umumiy
    foiz_stavkasi = 0.12
    hisoblar_soni = 0

    def __init__(self, egasi, boshlangich_balans=0):
        # Instance o'zgaruvchilar
        self.egasi = egasi
        self._balans = boshlangich_balans  # _ = "himoyalangan"
        self._operatsiyalar = []
        BankHisobi.hisoblar_soni += 1

    # Property — private o'zgaruvchiga xavfsiz kirish
    @property
    def balans(self):
        return self._balans

    def depozit(self, summa):
        if summa <= 0:
            raise ValueError("Summa musbat bo'lishi kerak")
        self._balans += summa
        self._operatsiyalar.append(f"+{summa:,.0f}")
        return self  # method chaining uchun

    def yechish(self, summa):
        if summa > self._balans:
            raise ValueError("Balans yetarli emas")
        self._balans -= summa
        self._operatsiyalar.append(f"-{summa:,.0f}")
        return self

    def oylik_foiz(self):
        foiz = self._balans * BankHisobi.foiz_stavkasi / 12
        self._balans += foiz
        return foiz

    # Dunder metodlar
    def __str__(self):
        return f"Hisob({self.egasi}: {self._balans:,.0f} so'm)"

    def __repr__(self):
        return f"BankHisobi('{self.egasi}', {self._balans})"

    def __lt__(self, other):  # < operatori
        return self._balans < other._balans


# Ishlatish
ali = BankHisobi("Ali", 1_000_000)
ali.depozit(500_000).depozit(200_000).yechish(100_000)  # chaining
print(ali)  # Hisob(Ali: 1,600,000 so'm)

import time, functools

# Taymer decorator
def taymer(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        t0 = time.perf_counter()
        natija = func(*args, **kwargs)
        print(f"⏱ {func.__name__}: {time.perf_counter()-t0:.4f}s")
        return natija
    return wrapper

# Retry decorator — xato bo'lsa qayta urinish
def retry(urinish=3):
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*a, **k):
            for i in range(urinish):
                try:
                    return func(*a, **k)
                except Exception as e:
                    print(f"Urinish {i+1}: {e}")
                    if i == urinish - 1:
                        raise
        return wrapper
    return decorator

import asyncio

async def ma'lumot_ol(id: int) -> dict:
    await asyncio.sleep(1)  # API so'rovini simulyatsiya
    return {"id": id, "ism": f"Foydalanuvchi {id}"}

async def asosiy():
    # Ketma-ket (3 soniya)
    for i in [1, 2, 3]:
        natija = await ma'lumot_ol(i)
        print(natija)

    # Parallel (1 soniya!) — gather bilan
    natijalar = await asyncio.gather(
        ma'lumot_ol(1),
        ma'lumot_ol(2),
        ma'lumot_ol(3)
    )
    for n in natijalar:
        print(n)

asyncio.run(asosiy())
